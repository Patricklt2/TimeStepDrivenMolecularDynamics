  \subsubsection{Definición Observables}
  \begin{frame}{Energía}
    \begin{itemize}
      \item Energía Potencial:
      \[E_{pot}(t) = -G \sum_{i<j} \frac{m_i m_j}{\sqrt{|r_{ij}(t)|^2+h^2}}\]

      \item Energía Cinética:
      \[E_{kin}(t) = \frac{1}{2} \sum_{i=1} m_i |v_i(t)|^2\]

      \item Energía Total:
      \[E_{tot}(t) = E_{kin}(t) + E_{pot}(t)\]
    \end{itemize}
  \end{frame}

  \begin{frame}{Conservación de la energía}
    \begin{itemize}
      \item Error Relativo:
      \[ \epsilon_{E}(t) = \frac{|E_{tot}(t) - E_{tot}(0)|}{|E_{tot}(0)|} \]
      \item Error relativo promedio:
      \[ \langle \epsilon_{E} \rangle = \frac{1}{t_{max}} \sum_{t=0}^{t_{max}} \epsilon_{E}(t) \]
    \end{itemize}
  \end{frame}

  \begin{frame}{Radio de Masa Media}
    \begin{itemize}
      \item Radio media masa:
      \[r_{hm} = mediana(|r_i(t)-r_{CM}(t)|) \] siendo:
      \[ r_{CM}(t) = \frac{1}{N} \sum_{i=1}^{N} r_i(t) \]
      \item Promedio del radio de media masa:
      \[ \langle r_{hm} \rangle = \frac{1}{M} \sum_{k=1}^{M} r_{hm}^{k}(t) \] donde M es la cantidad de simulaciones
    \end{itemize}
  \end{frame}

  \begin{frame}{Tiempo de Cruce}
    \begin{itemize}
      \item Tiempo de cruce:
      $t^*$ tal que $r_{hm}(t^*-dt) \leq 1$ AND $r_{hm}(t^*+dt) > 1$
      \item Promedio del tiempo de cruce:
      \[ \langle t^* \rangle = \frac{1}{M} \sum_{k=1}^{M} t_{k}^{*}(t) \] M cantidad de simulaciones
    \end{itemize}
  \end{frame}

  \begin{frame}{Conclusiones}
    \begin{itemize}
      \item A medida que $\Delta t$ incrementa, el desvío de la energía aumenta. El $\Delta t$ óptimo es el mayor valor para el cuál el desvío es mínimo.
      \item La energía total del sistema se conserva.
      \item Radio de masa media aumenta con el tiempo.
      \item Los sistemas con mayor número de partículas son más estables, ya que se expanden más lentamente (tardan más en superar el límite marcado por $r_{hm}=1$).
    \end{itemize}
